# Common mistakes, each planted in a copy of one program by
# scripts/compare_dx.sh, which then reports whether the copy builds, the
# compiler's first error if it does not, and otherwise what the copy answers to
# the request named. A block is
#
#   == NAME PROGRAM METHOD TARGET [BODY]
#   - a line of the program, matched with its indentation removed
#   + what replaces it, indented as the matched line
#
# Every `-` line must match at least one line, and every matching line is
# replaced. A POST body is sent with Content-Type: application/json. Each
# mistake is planted in both programs where the same intent can be written
# in both; a name that appears once has no counterpart in the other program
# (scripts/compare_dx.sh says so).

# The route's placeholder is renamed; the handler is not.
== renamed-placeholder muntin GET /users/2
- app.get["/users/{id}"](get_user, users)
+ app.get["/users/{user_id}"](get_user, users)
== renamed-placeholder flare GET /users/2
- r.get("/users/:id", Extracted(GetUser(users.copy(), PathInt["id"]())))
+ r.get("/users/:user_id", Extracted(GetUser(users.copy(), PathInt["id"]())))

# The handler's two route values are declared in the other order.
== swapped-values muntin GET /users/1/posts/7
- def get_post(uid: Param["uid", Int], pid: Param["pid", Int]) -> String:
+ def get_post(pid: Param["pid", Int], uid: Param["uid", Int]) -> String:
== swapped-values flare GET /users/1/posts/7
- var uid: PathInt["uid"]
+ var pid: PathInt["pid"]
- var pid: PathInt["pid"]
+ var uid: PathInt["uid"]

# The same, with the values declared without Param (docs/DX.md recommends
# Param for two values; bare values bind by position).
== swapped-bare-values muntin GET /users/1/posts/7
- def get_post(uid: Param["uid", Int], pid: Param["pid", Int]) -> String:
+ def get_post(pid: Int, uid: Int) -> String:
- return String("post ", pid.value, " of user ", uid.value)
+ return String("post ", pid, " of user ", uid)

# The handler reads a route value the route does not declare.
